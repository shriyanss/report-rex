% ── Fonts ──────────────────────────────────────────────────────────────────
% Scales land the defaults on whole half-points in PostScript points (bp, the
% unit Word uses; a TeX pt is 1/72.27in), so the `--docx` clone typesets and
% wraps the same: body 10bp (0.91667 x 10.95pt), inline code 9.5bp, code
% blocks 8.5bp (9.8pt x 0.87083), \large headings 11.5bp. Common
% ligatures (ff, fi) and kerning are off: Word sets neither by default.
% Line breaking is Word's too: spaces stretch (practically without limit) but
% never shrink, so every line that fits is equally good and TeX fills each
% line greedily, as Word does; lines break after a typed '-' (HyphenChar).
% The faces are Liberation Sans/Mono (SIL OFL, shipped in assets/fonts and
% embedded in the `--docx` clone), so both outputs draw the very same glyphs.
% They have no hook arrow, so continuation marks use → (Liberation's).
$if(mainfont)$
\setmainfont{$mainfont$}
$else$
\setmainfont{Liberation Sans}[Scale=0.91667, WordSpace={1,1000,0}, Ligatures={TeX,NoCommon}, Kerning=Off, HyphenChar=-]
$endif$
$if(monofont)$
\setmonofont{$monofont$}[Scale=MatchLowercase]
$else$
\setmonofont{Liberation Mono}[Scale=0.87083, Kerning=Off]
$endif$
% Display face for the cover title and section headings. Falls back to
% Liberation Sans; set `headingfont` in theme.yaml to a rounded
% geometric face (e.g. "TeX Gyre Adventor", "Poppins") for the source look.
$if(headingfont)$
\newfontfamily\headingfont{$headingfont$}
$else$
\newfontfamily\headingfont{Liberation Sans}[Scale=0.96193, WordSpace={1,1000,0}, Ligatures={TeX,NoCommon}, Kerning=Off, HyphenChar=-]
$endif$
